cmake_minimum_required(VERSION 3.24)
project(asa_server_plugin_advanced LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address/undefined diagnostics with GCC or Clang" OFF)

add_library(reward_plugin src/reward_plugin.cpp)
target_include_directories(reward_plugin PUBLIC include)
add_executable(asa_plugin_tests tests/main.cpp tests/lifecycle.cpp tests/economy.cpp
               tests/recovery.cpp tests/allocation_probe.cpp)
target_link_libraries(asa_plugin_tests PRIVATE reward_plugin)
foreach(TARGET_NAME reward_plugin asa_plugin_tests)
  if(MSVC)
    target_compile_options(${TARGET_NAME} PRIVATE /W4)
  else()
    target_compile_options(${TARGET_NAME} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(ENABLE_SANITIZERS)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
      target_compile_options(${TARGET_NAME} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${TARGET_NAME} PRIVATE -fsanitize=address,undefined)
    else()
      message(FATAL_ERROR "ENABLE_SANITIZERS requires a supported GCC/Clang runtime")
    endif()
  endif()
endforeach()

enable_testing()
file(STRINGS tests/cases.txt CASE_NAMES)
list(LENGTH CASE_NAMES CASE_COUNT)
set(UNIQUE_NAMES ${CASE_NAMES})
list(REMOVE_DUPLICATES UNIQUE_NAMES)
list(LENGTH UNIQUE_NAMES UNIQUE_COUNT)
if(NOT CASE_COUNT EQUAL 108 OR NOT UNIQUE_COUNT EQUAL 108)
  message(FATAL_ERROR "The protected inventory must contain exactly 108 unique cases")
endif()
foreach(CASE_NAME IN LISTS CASE_NAMES)
  add_test(NAME cpp.asa_plugin_advanced.${CASE_NAME} COMMAND asa_plugin_tests ${CASE_NAME})
  set_tests_properties(cpp.asa_plugin_advanced.${CASE_NAME} PROPERTIES TIMEOUT 20)
endforeach()
