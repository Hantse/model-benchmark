cmake_minimum_required(VERSION 3.24)
project(AsaServerPluginBenchmark VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(reward_plugin STATIC src/reward_plugin.cpp)
target_include_directories(reward_plugin PUBLIC include)
add_executable(asa_acceptance tests/acceptance.cpp)
target_link_libraries(asa_acceptance PRIVATE reward_plugin)
if(MSVC)
    target_compile_options(reward_plugin PRIVATE /W4)
    target_compile_options(asa_acceptance PRIVATE /W4)
else()
    target_compile_options(reward_plugin PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(asa_acceptance PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/tests/cases.txt" acceptance_cases)
foreach(case_name IN LISTS acceptance_cases)
    add_test(NAME "${case_name}" COMMAND asa_acceptance "${case_name}")
    set_tests_properties("${case_name}" PROPERTIES TIMEOUT 10)
endforeach()
