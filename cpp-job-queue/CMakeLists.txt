cmake_minimum_required(VERSION 3.24)
project(cpp_job_queue LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(job_queue src/job_queue.cpp)
target_include_directories(job_queue PUBLIC include)
add_executable(job_queue_tests tests/acceptance.cpp)
target_link_libraries(job_queue_tests PRIVATE job_queue)
if(MSVC)
  target_compile_options(job_queue PRIVATE /W4)
  target_compile_options(job_queue_tests PRIVATE /W4)
else()
  target_compile_options(job_queue PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(job_queue_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
file(STRINGS tests/cases.txt CASE_NAMES)
foreach(CASE_NAME IN LISTS CASE_NAMES)
  add_test(NAME cpp.job_queue.${CASE_NAME} COMMAND job_queue_tests ${CASE_NAME})
  set_tests_properties(cpp.job_queue.${CASE_NAME} PROPERTIES TIMEOUT 10)
endforeach()
